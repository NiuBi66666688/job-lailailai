快速开始

# 1) 获取
git clone https://github.com/NiuBi66666688/job-lailailai.git
cd job-lailailai

# 2) 验收（零依赖，不需要 pip install）
python verify.py

# 3) 装到本机 agent 平台
python scripts/install.py --detect
python scripts/install.py --target all --link

# 4) 直接生成一份引导页试试
python scripts/render.py examples/jobs.sample.json out.html --theme sci
